QUERIES SQL DE VALIDACAO
1) Usuario e role
SELECT id, email, role, membership_status, created_at
FROM users
WHERE email = 'qa.missionario@example.com';

2) Perfil missionary
SELECT m.id, m.user_id, m.status, m.missionary_agency_id, m.created_at
FROM missionaries m
JOIN users u ON u.id = m.user_id
WHERE u.email = 'qa.missionario@example.com';

3) Campos about/contatos/identidade
SELECT
  m.about_me,
  m.mission_story_summary,
  m.origin_location,
  m.prayer_request,
  m.life_verse,
  m.public_email,
  m.public_phone,
  m.donation_message,
  m.identity_type,
  m.identity_document
FROM missionaries m
JOIN users u ON u.id = m.user_id
WHERE u.email = 'qa.missionario@example.com';

4) Endereco de atuacao
SELECT mwa.id, mwa.missionary_id, mwa.zip, mwa.district, mwa.city, mwa.state, mwa.country
FROM missionary_work_addresses mwa
JOIN missionaries m ON m.id = mwa.missionary_id
JOIN users u ON u.id = m.user_id
WHERE u.email = 'qa.missionario@example.com';

5) Garantir relacao 1:1 no endereco
SELECT missionary_id, COUNT(*) AS total
FROM missionary_work_addresses
GROUP BY missionary_id
HAVING COUNT(*) > 1;

6) Sessoes/tokens
SELECT rt.id, rt.user_id, rt.family_id, rt.revoked_at, rt.expires_at, rt.device_type, rt.last_used_at
FROM refresh_tokens rt
JOIN users u ON u.id = rt.user_id
WHERE u.email = 'qa.missionario@example.com'
ORDER BY rt.created_at DESC;
